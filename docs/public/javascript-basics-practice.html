3. 函数定义演示
// 函数声明 function cultivate(technique) { return `修炼${technique}功法`; } // 函数表达式 const meditate = function(time) { return `冥想${time}分钟`; }; // 箭头函数 const breakthrough = (level) => `突破到${level}境界`; // 默认参数 function practice(technique = '基础功法', hours = 1) { return `修炼${technique}${hours}小时`; } console.log(cultivate('炼气诀')); console.log(meditate(30)); console.log(breakthrough('筑基期')); console.log(practice());

6. 循环和数组方法
// 数组操作演示 const techniques = ['炼气', '筑基', '金丹', '元婴']; // forEach方法 console.log('学习功法：'); techniques.forEach((tech, index) => { console.log(`${index + 1}. ${tech}功法`); }); // map方法 const levels = [1, 2, 3, 4, 5]; const powerLevels = levels.map(level => level * 10); console.log('战力等级：', powerLevels); // filter方法 const highLevels = levels.filter(level => level >= 3); console.log('高级等级：', highLevels);

8. 对象操作演示
// 对象创建和操作 let cultivator = { name: 'HOLIN', level: 5, techniques: ['炼气诀', '御剑术'], cultivate: function(hours) { return `${this.name}修炼了${hours}小时`; } }; // 对象解构 let {name, level, techniques} = cultivator; console.log('解构结果：', name, level, techniques); // 对象展开运算符 let newCultivator = { ...cultivator, level: 6, newTechnique: '阵法' }; console.log('新修仙者：', newCultivator); // 方法调用 console.log(cultivator.cultivate(2));
